\section{Mathematical Formulations}
\label{app:math}

\subsection{Pairwise Ranking Loss}

For a correct solution ($+$) and incorrect solution ($-$) from the same problem:
\begin{equation}
    \mathcal{L}_{\text{pair}} = -\log \sigma(\bar{V}^{+} - \bar{V}^{-})
\end{equation}
where $\bar{V} = \frac{1}{|\mathcal{O}|} \sum_{t \in \mathcal{O}} V_t$ averages over output tokens.

\subsection{Generalized Advantage Estimation}

TD-error (simplified for code generation with no intermediate rewards):
\begin{equation}
    \delta_t = \gamma V(s_{t+1}) - V(s_t)
\end{equation}

GAE accumulation:
\begin{equation}
    A_t = \sum_{l \geq 0} (\gamma\lambda)^l \delta_{t+l}
\end{equation}

With $\gamma = 1.0$ (no discounting) and $\lambda = 0.95$.

\subsection{Whitening Normalization}

\begin{equation}
    \hat{A}_t = \frac{A_t - \mu}{\sigma + \epsilon}
\end{equation}

where $\mu$ and $\sigma$ are EMA running statistics with momentum 0.1.

\subsection{Exponential Weighting with Clipping}

\begin{equation}
    w_t = \text{clip}\left(\exp\left(\frac{\hat{A}_t}{\beta}\right), w_{\min}, w_{\max}\right)
\end{equation}

With $\beta = 1.0$, $w_{\min} = 0.1$, $w_{\max} = 3.0$.

\subsection{WMTP Objective}

\begin{equation}
    \mathcal{L}_{\text{WMTP}} = \frac{1}{K} \sum_{k=1}^{K} \sum_t w_t \cdot \text{CE}(z_{t,k}, y_{t+k})
\end{equation}

\subsection{Pass@k Estimator}

Unbiased estimator from \citet{chen2021evaluating}:
\begin{equation}
    \text{Pass@}k = \mathbb{E}_{\text{problems}} \left[ 1 - \frac{\binom{n-c}{k}}{\binom{n}{k}} \right]
\end{equation}

where $n$ = samples per problem, $c$ = correct samples.

\section{Implementation Details}
\label{app:impl}

\subsection{Data Pipeline}

\begin{itemize}[nosep]
    \item \textbf{Template}: Alpaca format
    \begin{verbatim}
Below is an instruction that describes a task...
### Instruction:
{problem_description}
### Response:
{solution_code}
    \end{verbatim}

    \item \textbf{Loss masking}: Instruction and input tokens have \texttt{labels=-100} (ignored in loss)
    \item \textbf{Output tokens only}: Only solution code tokens contribute to loss
\end{itemize}

\subsection{Distributed Training}

\begin{itemize}[nosep]
    \item \textbf{FSDP}: \texttt{FULL\_SHARD} (ZeRO-3 equivalent)
    \item \textbf{Activation checkpointing}: Enabled for memory efficiency
    \item \textbf{Mixed precision}: BF16
    \item \textbf{EMA synchronization}: \texttt{all\_reduce} across ranks for consistent normalization
\end{itemize}

\subsection{Hardware}

\begin{itemize}[nosep]
    \item \textbf{GPUs}: NVIDIA H100 80GB $\times$ 4
    \item \textbf{Interconnect}: NVLink
    \item \textbf{Training time}: Approximately 4 hours per experiment
\end{itemize}

\section{LoRA Parameter Calculation}
\label{app:lora}

\subsection{Policy Model (Meta-LLaMA MTP 7B)}

\begin{itemize}[nosep]
    \item Hidden dimension: $d = 4096$
    \item LoRA rank: $r = 64$
    \item Target modules: wq, wk, wv, wo, w1, w2, w3 (7 modules per layer)
    \item Number of layers: $L = 32$
    \item Parameters per module: $2 \times d \times r = 2 \times 4096 \times 64 = 524,288$
    \item Total LoRA parameters: $7 \times 32 \times 524,288 = 117,440,512 \approx 117.4$M
    \item With MTP heads (4 heads): Additional parameters bring total to $\sim$190.8M
    \item Percentage of base: $190.8$M / $7$B = 2.73\%
\end{itemize}

\subsection{Critic Model (Sheared-LLaMA 2.7B)}

\begin{itemize}[nosep]
    \item Hidden dimension: $d = 2560$
    \item LoRA rank: $r = 32$
    \item Target modules: q\_proj, k\_proj, v\_proj, o\_proj, gate\_proj, up\_proj, down\_proj (7 modules)
    \item Number of layers: $L = 32$
    \item Parameters per module: $2 \times d \times r = 2 \times 2560 \times 32 = 163,840$
    \item Total LoRA parameters: $7 \times 32 \times 163,840 = 36,700,160 \approx 36.7$M
    \item With value head MLP: Additional $\sim$82.8M parameters
    \item Total trainable: $\sim$119.5M
    \item Percentage of base: $119.5$M / $2.7$B = 4.43\%
\end{itemize}
